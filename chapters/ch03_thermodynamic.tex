\section{Equation of State (EOS) of Stellar Matter}
\begin{intuition}
The equation of state (EOS) relates the pressure, density, and temperature of a substance. For stellar matter, the EOS is crucial for understanding the internal structure and evolution of stars.
\end{intuition}

\begin{theory}
\begin{align}
P &= P(\rho, T, \{X_i\}), \\    
\text{Ideal gas}: \quad P &= \frac{\rho k_B T}{\mu m_H}, \\
\text{Radiation pressure}: \quad P_{\text{rad}} &= \frac{1}{3}aT^4, \quad a = \frac{4\sigma}{c}, \\
\text{Degenerate gas}: \quad P_{\text{deg}} &\propto \rho^{5/3} \quad \text{(non-relativistic)}, \\
P_{\text{deg}} &\propto \rho^{4/3} \quad \text{(relativistic)}.
\end{align}
\end{theory}
