\section{Statistical Distributions}
\subsection{Energy-Momentum Relation}
To calculate the value of various thermodynamic quantities, we need first clarify some basic concepts and notations first for convenience. \\

In below contents, we will use the $E(p)$ notation to represent the energy of a particle with momentum $p$ and mass $m$.
\begin{foundation}
    Recall special relativity, the energy of a particle with mass $m$ and three-momentum $p$ is given by
    \[
        E(\vec{p}) = \sqrt{|\vec{p}|^2c^2 + m^2c^4}.
    \]
    Taken $c = 1$ and $p = |\vec{p}|$, we have $E(p) = \sqrt{p^2 + m^2}$. \\
\end{foundation}
\section{Statistical Distributions}

\begin{theory}
\begin{align}   
\text{Maxwell-Boltzmann}: \quad f_{\text{MB}}(E) &= \frac{2\pi}{(k_B T)^{3/2}} \sqrt{E} e^{-E/k_B T}, \\
\text{Fermi-Dirac}: \quad f_{\text{FD}}(E) &= \frac{1}{e^{(E-\mu)/k_B T} + 1}, \\
\text{Bose-Einstein}: \quad f_{\text{BE}}(E) &= \frac{1}{e^{(E-\mu)/k_B T} - 1}.
\end{align}
\end{theory}

\begin{example}{}{ex:mb}
Calculate the most probable energy $E_{\text{mp}}$ for a Maxwell-Boltzmann distribution.

\smallskip\textbf{Solution.}
The most probable energy is found by setting the derivative of the distribution function to zero:
\begin{equation}
\frac{d}{dE} f_{\text{MB}}(E) = 0.
\end{equation}
Solving this equation gives:
\begin{equation}
E_{\text{mp}} = \frac{k_B T}{2}.
\end{equation}
\end{example}

\section{Equation of State (EOS) of Stellar Matter}
\begin{intuition}
The equation of state (EOS) relates the pressure, density, and temperature of a substance. For stellar matter, the EOS is crucial for understanding the internal structure and evolution of stars.
\end{intuition}

\begin{theory}
\begin{align}
P &= P(\rho, T, \{X_i\}), \\    
\text{Ideal gas}: \quad P &= \frac{\rho k_B T}{\mu m_H}, \\
\text{Radiation pressure}: \quad P_{\text{rad}} &= \frac{1}{3}aT^4, \quad a = \frac{4\sigma}{c}, \\
\text{Degenerate gas}: \quad P_{\text{deg}} &\propto \rho^{5/3} \quad \text{(non-relativistic)}, \\
P_{\text{deg}} &\propto \rho^{4/3} \quad \text{(relativistic)}.
\end{align}
\end{theory}

\section{Radiation}
\begin{theory}
\begin{align}
u_{\text{rad}} &= aT^4, \\  
F_{\text{rad}} &= \frac{c}{4}u_{\text{rad}} = \sigma T^4, \\
L_{\text{rad}} &= 4\pi R^2 F_{\text{rad}} = 4\pi R^2 \sigma T^4.
\end{align} 
\end{theory}
