\subsection{Tolman-Oppenheimer-Volkoff (TOV) Equation}
\begin{theory}
The \textbf{Tolman-Oppenheimer-Volkoff (TOV) equation} describes the structure of a spherically symmetric body in hydrostatic equilibrium in general relativity. It is given by:
\begin{equation}
\frac{dP(r)}{dr} = -\frac{G \left(\rho(r) + \frac{P(r)}{c^2}\right) \left(m(r) + 4\pi r^3 \frac{P(r)}{c^2}\right)}{r^2 \left(1 - \frac{2Gm(r)}{rc^2}\right)}
\end{equation}
where $P(r)$ is the pressure, $\rho(r)$ is the energy density, $m(r)$ is the mass enclosed within radius $r$, $G$ is the gravitational constant, and $c$ is the speed of light. This equation is crucial for modeling neutron stars and other compact objects where relativistic effects are significant.
\end{theory}

\section{Black Holes}

\subsection{Schwarzschild Radius}
\begin{theory}
The \textbf{Schwarzschild radius} is the radius of the \textbf{event horizon} of a non-rotating black hole. It is given by:
\begin{equation}
r_s = \frac{2GM}{c^2}   
\end{equation}
where $G$ is the gravitational constant, $M$ is the mass of the black hole, and $c$ is the speed of light. Any object that is compressed within its Schwarzschild radius will become a black hole.  

\end{theory}

\begin{example}{Schwarzschild Radius of a Collapsing Star}{ex:schwarzschild_radius}
Suppose we have a star with a mass of $10 M_\odot$ (where $M_\odot$ is the mass of the Sun). Calculate the Schwarzschild radius of the black hole that would form if this star were to collapse.    

\smallskip\textbf{Solution.}
First, we need to convert the mass of the star into kilograms. The mass of the Sun is approximately $1.989 \times 10^{30}$ kg, so:
\begin{equation}
M = 10 M_\odot = 10 \times 1.989 \times 10^{30} \text{ kg} = 1.989 \times 10^{31} \text{ kg}
\end{equation}  
Using the formula for the Schwarzschild radius:
\begin{equation}
r_s = \frac{2GM}{c^2},
\end{equation}
we have
\begin{equation}
r_s = 29.49 \text{ km}
\end{equation}
\end{example}
